                             คู่มือคำสั่ง
                        

                        

                            

                                Declare
                                
ประกาศตัวแปร

                                
Declare Integer x, y
Declare String name

                            

                            

                                Assign
                                
กำหนดค่าให้ตัวแปร

                                
Assign x = 10

                            

                            

                                Input / Output
                                
รับค่า / แสดงผล

                                
Input x
Output "Hello"

                            

                            

                                If...Then...End If
                                
If x > 0 Then
    Output "Yes"
Else
    Output "No"
End If

                            

                            

                                For Loop
                                
For i = 1 To 10
    Output i
End For

                            

                            

                                While Loop
                                
While x < 5
    Assign x = x + 1
End While

                            

                            

                                Do...While Loop
                                
Do
    Assign x = x - 1
Loop x > 0

                            

                            

                                Array 1D / 2D
                                
Declare Integer arr[5]
Declare Integer m[2][2]
Assign arr[0] = 5
